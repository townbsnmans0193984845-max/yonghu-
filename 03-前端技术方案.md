### 路由守卫

```typescript
// src/router/AuthGuard.tsx
import { useEffect } from 'react';
import { useNavigate, useLocation } from 'react-router-dom';
import { useAuthStore } from '@/stores/authStore';

export const AuthGuard = ({ children }: { children: React.ReactNode }) => {
  const navigate = useNavigate();
  const location = useLocation();
  const { token, role } = useAuthStore();

  useEffect(() => {
    const requiresAuth = location.state?.requiresAuth;
    const requiredRole = location.state?.role;

    if (requiresAuth && !token) {
      navigate('/login', { replace: true });
      return;
    }

    if (requiredRole && role !== requiredRole) {
      navigate('/role-select', { replace: true });
    }
  }, [token, role, location, navigate]);

  return <>{children}</>;
};
```

## 状态管理

### Zustand Store设计

```typescript
// src/stores/authStore.ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

interface User {
  id: string;
  phone: string;
  role: 'worker' | 'recruiter' | null;
  name: string;
  credit_score: number;
}

interface AuthState {
  token: string | null;
  user: User | null;
  setAuth: (token: string, user: User) => void;
  clearAuth: () => void;
  updateUser: (user: Partial<User>) => void;
}

export const useAuthStore = create<AuthState>()(
  persist(
    (set) => ({
      token: null,
      user: null,
      setAuth: (token, user) => set({ token, user }),
      clearAuth: () => set({ token: null, user: null }),
      updateUser: (updates) =>
        set((state) => ({
          user: state.user ? { ...state.user, ...updates } : null,
        })),
    }),
    {
      name: 'auth-storage',
    }
  )
);
```

```typescript
// src/stores/jobStore.ts
import { create } from 'zustand';
import { Job } from '@/types/models';

interface JobState {
  jobs: Job[];
  filters: {
    keyword: string;
    city: string;
    salaryRange: [number, number];
  };
  setJobs: (jobs: Job[]) => void;
  setFilters: (filters: Partial<JobState['filters']>) => void;
  resetFilters: () => void;
}

export const useJobStore = create<JobState>((set) => ({
  jobs: [],
  filters: {
    keyword: '',
    city: '',
    salaryRange: [0, 1000],
  },
  setJobs: (jobs) => set({ jobs }),
  setFilters: (updates) =>
    set((state) => ({
      filters: { ...state.filters, ...updates },
    })),
  resetFilters: () =>
    set({
      filters: {
        keyword: '',
        city: '',
        salaryRange: [0, 1000],
      },
    }),
}));
```

## API请求封装

### Axios实例配置

```typescript
// src/services/api.ts
import axios, { AxiosError, AxiosRequestConfig } from 'axios';
import { useAuthStore } from '@/stores/authStore';

const BASE_URL = import.meta.env.VITE_API_BASE_URL || '/api/v1';

export const api = axios.create({
  baseURL: BASE_URL,
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json',
  },
});

// 请求拦截器
api.interceptors.request.use(
  (config) => {
    const { token } = useAuthStore.getState();
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

// 响应拦截器
api.interceptors.response.use(
  (response) => {
    return response.data;
  },
  (error: AxiosError<ApiResponse<never>>) => {
    const { response } = error;

    if (response?.status === 401) {
      // Token过期,清除登录状态
      useAuthStore.getState().clearAuth();
      window.location.href = '/login';
      return Promise.reject(error);
    }

    // 提取错误信息
    const message = response?.data?.message || '请求失败';
    return Promise.reject(new Error(message));
  }
);

// 通用请求方法
export const request = async <T = any>(
  config: AxiosRequestConfig
): Promise<ApiResponse<T>> => {
  try {
    return await api(config);
  } catch (error) {
    throw error;
  }
};
```

## 数据流设计

### React Query + Zustand 组合

```typescript
// src/hooks/useJobs.ts
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { jobService } from '@/services/jobs';
import { useJobStore } from '@/stores/jobStore';
import { message } from 'antd-mobile';

export const useJobs = () => {
  const queryClient = useQueryClient();
  const { filters } = useJobStore();

  // 获取岗位列表
  const {
    data: jobsData,
    isLoading,
    error,
  } = useQuery({
    queryKey: ['jobs', filters],
    queryFn: () => jobService.getJobs(filters),
    onSuccess: (data) => {
      useJobStore.getState().setJobs(data.data);
    },
  });

  // 创建岗位
  const createJobMutation = useMutation({
    mutationFn: jobService.createJob,
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['jobs'] });
      message.success('发布成功');
    },
    onError: (error: Error) => {
      message.error(error.message);
    },
  });

  return {
    jobs: jobsData?.data || [],
    total: jobsData?.total || 0,
    isLoading,
    error,
    createJob: createJobMutation.mutate,
    isCreating: createJobMutation.isPending,
  };
};
```
